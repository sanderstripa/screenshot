cmake_minimum_required(VERSION 3.21)
project(Screenshot VERSION 0.1.0 LANGUAGES CXX RC)
add_executable(Screenshot WIN32 src/main.cpp assets/screenshot.rc)
target_compile_features(Screenshot PRIVATE cxx_std_17)
target_compile_definitions(Screenshot PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(Screenshot PRIVATE user32 gdi32 d2d1 dwmapi windowscodecs ole32)
if(MSVC)
  target_compile_options(Screenshot PRIVATE /W4 /EHsc /O2 /utf-8)
  target_link_options(Screenshot PRIVATE /OPT:REF /OPT:ICF)
endif()
